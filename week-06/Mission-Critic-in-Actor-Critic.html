import gymnasium as gym
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim

from torch.distributions import Categorical
import matplotlib.pyplot as plt

from safetensors.torch import save_file, load_file
from pathlib import Path

# List of few hyperparameters
gamma = 0.99
critic_weight = 0.5
num_episodes = 5000 # Run at least 20 episodes please
learning_rate = 0.001

model_name = f"ActorCritic-lr={learning_rate}+ep={num_episodes}"
model_save_dir = f"./saved_models/"
model_file = f"{model_save_dir}/{model_name}.safetensors"

try:
    # Creates the directory; ignores the error if it already exists
    Path(model_save_dir).mkdir(parents=True, exist_ok=True)
except:
    print(f'Error: mkdir {model_save_dir} failed.')

# Reproducibility seed
seed = 42

torch.manual_seed(seed)
np.random.seed(seed)


# Environment
env = gym.make("CartPole-v1")

state_dim = env.observation_space.shape[0]
action_dim = env.action_space.n


# Actor-Critic Network
class ActorCritic(nn.Module):

    def __init__(self, state_dim, action_dim):
        super().__init__()

        self.shared = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU()
        )

        self.actor = nn.Linear(
            128,
            action_dim
        )

        self.critic = nn.Linear(
            128,
            1
        )


    def forward(self, state):

        x = self.shared(state)

        logits = self.actor(x)

        value = self.critic(x)

        return logits, value


# Create the model object
model = ActorCritic(
    state_dim,
    action_dim
)

# Select gradient descent optimizer variant: Adam optimizer
optimizer = optim.Adam(
    model.parameters(),
    lr=learning_rate
)



# Action Selection
def select_action(state):

    state_tensor = torch.tensor(
        state,
        dtype=torch.float32
    ).unsqueeze(0)

    logits, value = model(state_tensor)

    distribution = Categorical(
        logits=logits
    )

    action = distribution.sample()

    log_prob = distribution.log_prob(
        action
    )

    return (
        action.item(),
        log_prob.squeeze(),
        value.squeeze()
    )

# Training loop that runs for `num_episodes`
episode_rewards = []

for episode in range(num_episodes):

    if episode == 0:
        state, info = env.reset(seed=seed)
    else:
        state, info = env.reset()

    total_reward = 0

    while True:

        # Actor chooses action
        action, log_prob, value = \
            select_action(state)


        # Environment transition
        next_state, reward, \
        terminated, truncated, info = \
            env.step(action)

        done = terminated or truncated


        # Critic evaluates next state
        next_state_tensor = torch.tensor(
            next_state,
            dtype=torch.float32
        ).unsqueeze(0)

        with torch.no_grad():

            _, next_value = model(
                next_state_tensor
            )

            next_value = \
                next_value.squeeze()


        # No future reward after TRUE termination
        if terminated:
            next_value = torch.tensor(0.0)


        # TD Target
        reward_tensor = torch.tensor(
            reward,
            dtype=torch.float32
        )

        td_target = (
            reward_tensor
            + gamma * next_value
        )


        # TD Error
        td_error = (
            td_target - value
        )


        # Actor Loss
        actor_loss = (
            -log_prob
            * td_error.detach()
        )


        # Critic Loss

        critic_loss = \
            td_error.pow(2)


        # Total Loss

        loss = (
            actor_loss
            + critic_weight
            * critic_loss
        )


        # Backpropagation
        optimizer.zero_grad()

        loss.backward()

        optimizer.step()


        # Move forward
        state = next_state

        total_reward += reward


        if done:
            break


    episode_rewards.append(
        total_reward
    )


    if (episode + 1) % 20 == 0:

        avg_reward = np.mean(
            episode_rewards[-20:]
        )

        print(
            f"Episode {episode+1:4d} | "
            f"Reward: {total_reward:6.1f} | "
            f"Average(20): {avg_reward:6.1f}"
        )

print('Training completed.')


env.close()


# Now, save the trained agent's model weights
save_file(model.state_dict(), f"{model_file}")

1.1 There are issues in the actor-critic paradise!

Current Actor-critic implementation trained for 600 episodes

Current Actor-critic implementation trained for 1000 episodes

Current Actor-critic implementation trained for 1500 episodes

and,

The agent learns well and then collapses around 800+ episodes

We can diagnose the earlier code as unstable rather than conclude that Actor–Critic is inherently performing poorly.

1.1.1 Issue A: The shared actor/critic network is the biggest concern

Recall that we used:

self.shared = nn.Sequential(...)
self.actor = nn.Linear(...)
self.critic = nn.Linear(...)

and then one loss:

\[ L=L_{\text{actor}}+0.5L_{\text{critic}}. \]

This means both actor and critic modify the same hidden-layer parameters.

Now imagine a well-trained state for which:

\[ V(s_t)=80. \]

The agent makes one bad stochastic action and terminates. Since CartPole-v1 gives \(+1\) on the termination step,

\[ \text{TD target}=1 \]

and therefore

\[ \delta_t=1-80=-79. \]

The critic loss becomes

\[ L_C=(-79)^2=6241. \]

Even multiplied by \(0.5\), i.e., the critic’s weight, this can dominate the actor loss. That very large gradient modifies the shared representation used by the actor, potentially changing a previously good policy dramatically.

This can repeat and produce:

\[ \text{good policy} \rightarrow \text{rare bad episode} \rightarrow \text{large critic error} \rightarrow \text{large shared-network update} \rightarrow \text{worse policy} \rightarrow \text{more bad episodes} \rightarrow \boxed{\text{collapse}} \]

That fits our observation about the collapsed learning outcome past 800 episodes.

1.1.2. Issue B: Learning rate 0.001 is actually fairly aggressive here

Our algorithm performs an optimizer update at every environment step.

Suppose the agent averages 300 steps per episode. Then 800 episodes could produce roughly \(800\times300=240,000\) gradient updates.

So lr = 0.001 is not particularly conservative in this setting.

Compare that with REINFORCE, where you might make only one optimization update per episode.

This is an important difference:

\[ 800\text{ Actor–Critic episodes} \neq 800\text{ REINFORCE updates}. \]

1.1.3. Issue C: The stochastic actor can destroy itself after it has learned

Suppose, the actor eventually develops

\[ \pi(a|s)=[0.98,0.02]. \]

It still occasionally samples the \(0.02\) action.

  • If that action produces a large negative TD error, the resulting updates can be substantial.
    • Without entropy control, gradient clipping, batching, or other stabilizers, the network can drift.

RL learning curves are therefore not guaranteed to improve monotonically.

1.3 Use Huber loss for the critic

Instead of:

critic_loss = td_error.pow(2)

Let’s now use use:

critic_loss = torch.nn.functional.huber_loss(
    value,
    td_target.detach()
)

Why?

With ordinary MSE,

\[ \delta=-80 \]

produces

\[ \delta^2=6400. \]

Huber loss is much less dominated by large occasional TD errors.

That is particularly useful during early training and occasional catastrophic transitions.

1.3.1 The Huber Loss

In an Actor–Critic algorithm, the critic is trying to answer:

“How good is this state?”

It predicts a value

\[ V_\phi(s_t) \]

and we give it a learning target, typically the one-step TD target:

\[ y_t = r_t + \gamma V_\phi(s_{t+1}). \]

So the critic’s prediction error is

\[ \delta_t = y_t - V_\phi(s_t). \]

This is the same TD error you have already been using. The critic needs a loss function that says, “How bad was this prediction?”

A natural choice we took last time is the squared TD delta error:

\[ L_{\text{MSE}} = \delta_t^2. \]

But, Huber loss is often more forgiving and stable than the squared loss we discussed briefly a moment ago.

1.3.2 Intuition: Huber loss = MSE for small mistakes, MAE for big mistakes

Huber loss behaves like this:

\[ L_\delta(e)= \begin{cases} \frac{1}{2}e^2, & |e| \le \delta \\[6pt] \delta\left(|e|-\frac{1}{2}\delta\right), & |e|>\delta \end{cases} \]

where

\[ e = y_t - V(s_t). \]

Usually we can think of \(\delta=1\).

So:

  • small critic error → square it
  • large critic error → grow approximately linearly

That second part is important in RL because TD targets can occasionally be quite noisy, and an example scenario we already discussed in section 1.3. Let’s talk about it again.

1.3.3 Plot Huber Loss

import numpy as np
import matplotlib.pyplot as plt

# Define the Huber Loss formula
def huber_loss(r, delta=1.5):
    r_abs = np.abs(r)
    return np.where(r_abs <= delta, 0.5 * r**2, delta * (r_abs - 0.5 * delta))

# Generate residual values (errors)
r = np.linspace(-5, 5, 500)

plt.figure(figsize=(8, 5))

# Plot reference loss functions
plt.plot(r, 0.5 * r**2, label='Squared Error (L2 / MSE)', color='gray', linestyle='--')
plt.plot(r, np.abs(r), label='Absolute Error (L1 / MAE)', color='black', linestyle=':')

# Plot Huber Loss with different delta thresholds
for delta in [1.0, 2.0]:
    plt.plot(r, huber_loss(r, delta=delta), label=f'Huber Loss (delta={delta})', lw=2)

# Graph configurations
plt.title('Huber Loss vs. L1 and L2 Metrics')

1.3.4 Easy example

Suppose the critic says

\[ V(s_t)=5 \]

but after observing the reward and next state we calculate

\[ r_t+\gamma V(s_{t+1})=6. \]

Therefore,

\[ e = 6-5=1. \]

This is a small mistake.

With Huber loss:

\[ L=\frac12(1)^2=0.5. \]

The critic gets a normal quadratic penalty.

Now suppose something unusual happens and the target is

\[ y_t=15. \]

The critic still predicted \(5\), so

\[ e=15-5=10. \]

With MSE:

\[ L_{\text{MSE}} = 10^2=100. \]

That is a huge penalty from just one transition.

With Huber loss, using \(\delta=1\):

\[ L_{\text{Huber}} = 1(10-\frac12) = 9.5. \]

So instead of suddenly getting a loss of 100, we get 9.5 with Huber-loss.

That’s essentially the reason Huber loss is useful for the critic:

Don’t let one surprisingly bad value estimate produce an enormous gradient and destabilize learning.

1.3.5 Connecting this directly to Actor–Critic

Suppose we have:

\[ r_t = 1 \]

\[ \gamma = 0.99 \]

and

\[ V(s_{t+1})=8. \]

Then the critic’s target is \(y_t = 1 + 0.99(8) = 8.92\)

Suppose the critic currently predicts, \(V(s_t)=6\)

Then

\[ \delta_t = 8.92 - 6 = 2.92. \]

This TD error actually plays two roles in a simple Actor–Critic algorithm:

  • For the actor: it can serve as the advantage: \(A_t \approx \delta_t\),
    • Since, \(\delta_t>0\), the action was better than the critic expected, so the actor should make that action more likely.
  • For the critic: we ask:

How much should we penalize its inaccurate prediction?

We might use

\[ L_{\text{critic}} = \text{Huber}(V(s_t),y_t). \]

Since \(2.92>1\),

\[ L = 2.92-\frac12 = 2.42. \]

Notice the conceptual separation:

\[ \boxed{ \delta_t = r_t+\gamma V(s_{t+1})-V(s_t) } \]

tells the actor

“Was the action better or worse than expected?”

while

\[ \boxed{ L_{\text{critic}} = \text{Huber}\left( V(s_t), r_t+\gamma V(s_{t+1}) \right) } \]

tells the critic

“Adjust your value prediction toward the TD target.”

1.3.9 So, are we still talking about Huber loss, huh!

In a summary:

Huber loss tells the critic: fix small mistakes precisely, but don’t overreact to very large mistakes.

That is particularly useful in reinforcement learning because the critic’s target is itself an estimate, so occasional large TD errors are quite okay.

1.4 Add gradient clipping

This is probably the simplest stabilization mechanism:

  • scales down the gradients of a neural network in-place if their collective norm exceeds a specific threshold.
  • It is primarily used to solve the exploding gradient problem, which is common when training architectures like Recurrent Neural Networks (RNNs) and Transformers
  • What it does:
    • It computes the total \(L_{p}\) norm (usually type of norm, \(p=2\), i.e., \(L_2\) norm) across all network gradients.
    • Then, if this total norm is greater than the max_norm value you set, it scales all gradients down proportionally.
    • Because every gradient is scaled by the exact same ratio, the direction of the gradient vector in high-dimensional space remains identical, but its magnitude is shrunk to fit safety bounds.
actor_optimizer.zero_grad()
actor_loss.backward()

torch.nn.utils.clip_grad_norm_(
    actor.parameters(),
    norm_type=2.0, #L-2 norm here
    max_norm=0.5 #The maximum allowed norm for the combined gradients.
)

actor_optimizer.step()

Do the same for the critic:

critic_optimizer.zero_grad()
critic_loss.backward()

torch.nn.utils.clip_grad_norm_(
    critic.parameters(),
    norm_type=2.0, #L-2 norm here
    max_norm=0.5 #The maximum allowed norm for the combined gradients.
)

critic_optimizer.step()

The idea is simple:

“A surprising experience can influence the network, but we don’t allow one experience to completely rewrite it.”

1.5 Add a small entropy bonus

Compute:

distribution = Categorical(logits=logits)

action = distribution.sample()

log_prob = distribution.log_prob(action)

entropy = distribution.entropy()

Here,

The entropy, also known as Shannon Entropy, of a probability distribution measures the average level of uncertainty, surprise, or randomness in a random variable’s possible outcomes.

For instance, for a discrete random variable \(X\) with possible outcomes \(x_{i}\) and probability distribution \(p(x_i)\), the Shannon entropy \(H(X)\) is defined as

\[ H(X) = -\sum_{i}_{i=1}^n p(X_{i}) \log p(X_{i}) \]

  • If base of the \(\log\) is 2, the unit of entropy is in bits.
  • If base of the \(\log\) is \(e\), the unit of entropy is in nats.

Two extreme situations:

  • Minimum uncertainty: if one outcome has a probability of \(1\) and all others are \(0\),

Say, out of \(n\) outcomes, \(p(X_{2})=1\) while all other outcomes are impossible. \[ H(X) = p(X_{1})\log p(X_{1}) + p(X_{2}) \log p(X_{2}) + \cdots + p(X_{n}) \log p(X_{n}) = 0 \]

  • Maximum uncertainty: if \(n\) outcomes are equally likely, i.e., uniform distribution having \(p(X_{i})=\frac{1}{n}\), the entropy reaches its maximum value of \(\log_{2}n\) bits.

\[ H(X) = -\sum_{i} \dfrac{1}{n} \log \left( \dfrac{1}{n}\right) = \log_2 n \]

Then:

entropy_beta = 0.001

actor_loss = (
    -log_prob * td_error.detach()
    - entropy_beta * entropy
)

Minimizing the actor_loss means you are trying to make this value as negative as possible.Making a negative entropy term smaller requires maximizing the entropy value itself.

This discourages the policy from becoming excessively certain too early.

The entropy term in the loss function forces the optimizer to maximize entropy, which keeps the policy’s action probabilities spread out instead of sharply focused on a single choice.

Conceptually:

\[ L_{\text{actor}} = -\log\pi(a_t|s_t)\delta_t - \beta H(\pi) \]

where

\[ H(\pi) \]

is policy entropy.

So, this strategy forces Exploration.

By rewarding high entropy, the agent is incentivized to keep trying alternative actions rather than prematurely committing to the very first strategy that yields a positive reward.

1.8 One more thing: if the return is literally 0

With default current CartPole-v1, every step receives \(+1\), including the termination step. Thus a raw episode return of exactly zero should not normally occur.

If you literally see zero rather than “very close to the bottom of the plot,” add:

print(episode, total_reward)

before appending it:

episode_rewards.append(total_reward)

because that could indicate a separate bookkeeping/plotting issue.

1.9 Do not train indefinitely once it is good

Suppose the moving average reaches:

\[ 450,\;470,\;490,\;495 \]

and then you continue training for another 1,000 episodes.

There is no rule saying:

1.10 Lesson learned

\[ \boxed{ \text{Actor–Critic is not simply “REINFORCE but better.”} } \]

The critic reduces one problem of REINFORCE, the high-variance Monte Carlo feedback, but in turn introduces another:

\[ \boxed{\text{We are now dependent on the quality and stability of the critic.}} \]

class Actor(nn.Module):
    def __init__(self, state_dim, action_dim):
        super().__init__()

        self.net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),

    def __init__(self, state_dim):
        super().__init__()

# Action Selection by the Actor
def select_action(state):

    state_tensor = torch.tensor(
        state,
        dtype=torch.float32
    ).unsqueeze(0)

    #please note, we are using the actor model now
    logits = actor(state_tensor)

    distribution = Categorical(
        logits=logits
    )

    action = distribution.sample()

    log_prob = distribution.log_prob(
        action
    )
    entropy = distribution.entropy()

    return (
        action.item(),
        log_prob.squeeze(),
        entropy.squeeze()
    )

# Training loop that runs for `num_episodes`
episode_rewards = []

for episode in range(num_episodes):

    episode_entropies = []
    episode_td_errors = []
    episode_values = []

    if episode == 0:
        state, info = env.reset(seed=seed)
    else:
        state, info = env.reset()
    
    #convert state into pytorch tensor
    state_tensor = torch.tensor(state,dtype=torch.float32).unsqueeze(0)
    

    total_reward = 0

    while True:
        # Actor chooses action based on the state it is in now
        # Please note, the return values. We are receiving entropy too
        action, log_prob, entropy = \
            select_action(state)


        # Environment transition
        next_state, reward, \
        terminated, truncated, info = \
            env.step(action)

        done = terminated or truncated

        #current state value as measured by our beloved `critic`
        value = critic(state_tensor).squeeze()

        # Critic evaluates next state
        next_state_tensor = torch.tensor(next_state,dtype=torch.float32).unsqueeze(0)
        with torch.no_grad():
            next_value = critic(next_state_tensor)
            next_value = next_value.squeeze()


        # No future reward after TRUE termination
        if terminated:
            next_value = torch.tensor(0.0)


        # TD Target
        reward_tensor = torch.tensor(reward,dtype=torch.float32)
        td_target = (reward_tensor + gamma * next_value)


        # TD Error, i.e., advantage
        td_error = ( td_target - value)


        # Update critic
        critic_loss = torch.nn.functional.huber_loss(value,td_target.detach())
        critic_optimizer.zero_grad()
        critic_loss.backward()
        torch.nn.utils.clip_grad_norm_(
            critic.parameters(),
            norm_type=2,
            max_norm=0.5
            )
        critic_optimizer.step()

        # Update actor
        actor_loss = (
            -log_prob * td_error.detach()
            - entropy_beta * entropy
        )
        actor_optimizer.zero_grad()
        actor_loss.backward()
        torch.nn.utils.clip_grad_norm_(
            actor.parameters(),
            norm_type=2,
            max_norm=0.5
        )
        actor_optimizer.step()


        # Enough updating!!!
        # Now, let's move forward with next state transition
        state = next_state
        state_tensor = next_state_tensor

        # aggregating rewards received (for accounting purposes. Yes, you heard it right.)
        total_reward += reward


        episode_entropies.append(entropy.item())
        episode_td_errors.append(abs(td_error.item()))
        episode_values.append(value.item())


        if done:
            break

        total_reward
    )

            f"Episode: {episode+1:4d} | "

        )

3. 1-step Actor-Critic vs \(n\)-step Actor-Critic

Let’s learn something new.

  • A simple analogy:
    • stochastic gradient descent vs mini-batch gradient descent (with batch-size \(n\))

3.1 From 1-step Actor-Critic to \(n\)-step Actor-Critic

An important change we can think of is to replace the current one-step TD target + optimizer update after every transition with a short \(n\)-step rollout. Say, we do 5-step rollout now.

3.2 The idea: 5-step Actor–Critic

Currently you use

\[ G_t^{(1)} = r_t+\gamma V(s_{t+1}). \]

Instead, collect up to 5 transitions and use

\[ \boxed{ G_t^{(5)} = r_t+\gamma r_{t+1} +\gamma^2r_{t+2} +\gamma^3r_{t+3} +\gamma^4r_{t+4} +\gamma^5V(s_{t+5}) } \]

Then the advantage becomes

\[ \boxed{ A_t=G_t^{(5)}-V(s_t) } \]

This gives the actor more meaningful feedback than a single reward while still avoiding the full-episode wait of REINFORCE.

The progression is thus beautifully intuitive:

\[ \text{1-step TD} \quad\longrightarrow\quad \text{5-step return} \quad\longrightarrow\quad \text{full Monte Carlo return}. \]

3.3 Subtle change in storing the 5-step info

Rather than storing log_prob tensors during the five environment interactions, we are storing only:

rollout_states
rollout_actions
rollout_rewards

Then, when the rollout is finished, we run the Actor again:

logits = actor(states_tensor)

distribution = Categorical(logits=logits)

log_probs = distribution.log_prob(actions_tensor)

That keeps the computational graph clean.

You can see the two distinct phases:

COLLECT EXPERIENCE
      ↓
[s, a, r]
[s, a, r]
[s, a, r]
[s, a, r]
[s, a, r]
      ↓
COMPUTE RETURNS
      ↓
UPDATE NETWORKS

This is closer to how modern Actor–Critic implementations are organized.

3.4 What exactly is happening to the 5-step return?

Suppose the rollout contains rewards

\[ r_0,r_1,r_2,r_3,r_4 \]

and the final state has

\[ V(s_5)=20. \]

For the first state:

\[ G_0^{(5)} = r_0+ \gamma r_1+ \gamma^2r_2+ \gamma^3r_3+ \gamma^4r_4+ \gamma^5(20). \]

But for the third state we don’t throw away the same rollout:

\[ G_2^{(3)} = r_2+ \gamma r_3+ \gamma^2r_4+ \gamma^3V(s_5). \]

This is why the backward computation is convenient:

G = bootstrap_value

for reward in reversed(rollout_rewards):

    G = reward + gamma * G

    returns.insert(0, G)

For example:

bootstrap = V(s5)

G4 = r4 + γV(s5)

G3 = r3 + γG4

G2 = r2 + γG3

G1 = r1 + γG2

G0 = r0 + γG1

So with just a few lines, every transition gets an appropriate multi-step return.

3.5 The most important conceptual change

The one-step Actor-Critic computes:

\[ \delta_t = r_t+\gamma V(s_{t+1})-V(s_t) \]

and immediately updates the actor.

The new version, i.e., the n-step Actor-critic instead computes:

\[ \boxed{ A_t = G_t^{(n)} - V(s_t) } \]

where \(G_t^{(n)}\) contains several actual rewards before asking the critic to estimate the remainder.

This is the important bias/variance bridge:

1-step Actor-Critic
       |
       | mostly critic estimates
       v
5-step Actor-Critic
       |
       | more real rewards
       v
REINFORCE
       |
       | complete real episode return
       v
Monte Carlo

The farther below we move, the less we rely on bootstrapping, but generally the more variance we introduce.

# Actor: our separate actor network
class Actor(nn.Module):

    def __init__(self, state_dim, action_dim):

        super().__init__()

        self.net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),

            nn.Linear(128, action_dim)
        )


    def forward(self, state):

        return self.net(state)


# Critic -- a separate critic network
class Critic(nn.Module):

    def __init__(self, state_dim):

        super().__init__()

        self.net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),

            nn.Linear(128, 1)
        )


    def forward(self, state):

        return self.net(state)


# Create two separate models, one for each networks we defined above
actor = Actor(state_dim,action_dim)

critic = Critic(state_dim)


# We have two Separate optimizers, one for each networks we defined
actor_optimizer = optim.Adam(
    actor.parameters(),
    lr=actor_learning_rate
)

critic_optimizer = optim.Adam(
    critic.parameters(),
    lr=critic_learning_rate

    if episode == 0:

    else:
        state, info = env.reset()


    done = False
    total_reward = 0

    #few episode specific lists,
    # again these are for logging
    episode_entropies = []
    episode_advantages = []
    episode_critic_losses = []


    # One episode loop
    while not done:

        # Store a SHORT rollout
        rollout_states = []
        rollout_actions = []
        rollout_rewards = []

        # Collect up to n_steps transitions
        for step in range(n_steps):

            state_tensor = torch.tensor(state,dtype=torch.float32).unsqueeze(0)


            # We only need an action during data collection.
            # Gradients will be calculated later for the batch.
            with torch.no_grad():
                logits = actor(state_tensor)

                distribution = Categorical(logits=logits)

                action = distribution.sample()


            # Take action

            next_state, reward, \
            terminated, truncated, info = \
                env.step(action.item())


            done = terminated or truncated

            # Store transition
            rollout_states.append(state)

            rollout_actions.append(action.item())

            rollout_rewards.append(reward)


            total_reward += reward

            state = next_state


            if done:
                break


        # BOOTSTRAP
        #
        # If the MDP truly terminated:
        #        future value = 0
        # Otherwise:
        #        future value = V(s_next)
        #

        if terminated:
            bootstrap_value = 0.0
        else:
            next_state_tensor = torch.tensor(
                state,
                dtype=torch.float32
            ).unsqueeze(0)

            with torch.no_grad():
                bootstrap_value = critic(next_state_tensor).item()


        # Calculate n-step returns BACKWARD
        #
        # Start with:
        #
        #       V(s_{t+n})
        #
        # Then move backward:
        #
        #       G = r + gamma * G
        #

        returns = []

        G = bootstrap_value

        for reward in reversed(rollout_rewards):
            G = reward + gamma * G
            returns.insert(0,G)


        # Convert rollout to tensors
        states_tensor = torch.tensor(
            np.array(rollout_states),
            dtype=torch.float32
        )

        actions_tensor = torch.tensor(
            rollout_actions,
            dtype=torch.long
        )

        returns_tensor = torch.tensor(
            returns,
            dtype=torch.float32
        )


        # ACTOR calculations
        logits = actor(states_tensor)

        distribution = Categorical(logits=logits)

        log_probs = distribution.log_prob(actions_tensor)

        entropy = distribution.entropy()


        # CRITIC calculations
        values = critic(states_tensor).squeeze(-1)


        # Advantage
        #
        # A_t = ( n-step return ) - (critic prediction )
        #

        advantages = ( returns_tensor - values.detach() )

        # Actor loss
        actor_loss = (
            -(log_probs * advantages).mean()
            - entropy_beta * entropy.mean()
        )

        # Critic loss is Explicit Huber loss
        critic_loss = F.huber_loss(
            values,
            returns_tensor
        )


        # Update Actor
        actor_optimizer.zero_grad()
        actor_loss.backward()
        actor_optimizer.step()


        # Update Critic
        critic_optimizer.zero_grad()
        critic_loss.backward()
        critic_optimizer.step()


        # Gather the diagnostics info

        episode_entropies.append(
            entropy.mean().item()
        )

        episode_advantages.append(
            advantages.abs().mean().item()
        )

        episode_critic_losses.append(
            critic_loss.item()
        )

    episode_rewards.append(total_reward)

For two actions, maximum entropy is approximately

\[ \log 2\approx0.693. \]

You do not necessarily want entropy to remain at \(0.693\). As the policy learns, it should generally fall. What you want to avoid is:

\[ 0.69 \rightarrow 0.30 \rightarrow 0.05 \rightarrow 0.001 \]

within the first few dozen episodes while performance remains poor.

That would indicate premature policy collapse.

The \(n\)-step Actor Critic summarizes like:

\[ \boxed{ \text{Advantage} = \text{n-step return} - \text{critic prediction} } \]

we implemented this with the following:
